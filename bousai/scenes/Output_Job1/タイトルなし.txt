    <script>
        // API Keyは実行環境から提供されます
        const apiKey = ""; 
        const apiUrl = `https://generativelanguage.googleapis.com/v1beta/models/imagen-4.0-generate-001:predict?key=${apiKey}`;

        // DOM要素の取得
        const globalPromptInput = document.getElementById('globalPrompt');
        const imageInput = document.getElementById('imageInput');
        const fileInput = document.getElementById('fileInput');
        const generateBtn = document.getElementById('generateBtn');
        const generateEnhancedBtn = document.getElementById('generateEnhancedBtn');
        const downloadZipBtn = document.getElementById('downloadZipBtn');
        const statusContainer = document.getElementById('statusContainer');
        const statusText = document.getElementById('statusText');
        const loader = document.getElementById('loader');
        const gallery = document.getElementById('gallery');

        let generatedZip = new JSZip();
        let hasGeneratedImages = false;

        // Gemini LLM API (テキスト生成用)
        const geminiApiUrl = `https://generativelanguage.googleapis.com/v1beta/models/gemini-2.5-flash-preview-09-2025:generateContent?key=${apiKey}`;

        /**
         * プロンプトをGemini APIで詳細に拡張する関数
         */
        async function enhancePrompt(prompt) {
            const delays = [1000, 2000, 4000, 8000, 16000];
            let attempt = 0;

            const payload = {
                contents: [{ parts: [{ text: `以下の短いアイデアを、画像生成AI向けの非常に詳細で美しい英語のプロンプトに拡張してください。構図、光、スタイル、雰囲気などの詳細を含めてください。出力は英語のプロンプトのテキストのみとし、前置きなどは不要です。\n\nアイデア: ${prompt}` }] }],
                systemInstruction: { parts: [{ text: "You are an expert prompt engineer for text-to-image models. Output ONLY the highly detailed English prompt." }] }
            };

            while (attempt < 5) {
                try {
                    const response = await fetch(geminiApiUrl, {
                        method: 'POST',
                        headers: { 'Content-Type': 'application/json' },
                        body: JSON.stringify(payload)
                    });
                    if (!response.ok) throw new Error(`HTTP error! status: ${response.status}`);
                    const data = await response.json();
                    
                    if (data.candidates && data.candidates[0].content.parts[0].text) {
                        return data.candidates[0].content.parts[0].text.trim();
                    } else {
                        throw new Error('Failed to parse Gemini response');
                    }
                } catch (error) {
                    if (attempt === 4) return prompt; // 失敗時は元のプロンプトをそのまま使用
                    await new Promise(resolve => setTimeout(resolve, delays[attempt]));
                    attempt++;
                }
            }
            return prompt;
        }

        /**
         * 画像を16:9の中央クロップでトリミングする関数
         * @param {string} base64ImageUrl 元画像のDataURL
         * @returns {Promise<string>} 16:9にクロップされたDataURL
         */
        function cropImageTo16x9(base64ImageUrl) {
            return new Promise((resolve, reject) => {
                const img = new Image();
                img.onload = () => {
                    const canvas = document.createElement('canvas');
                    const ctx = canvas.getContext('2d');

                    // 元画像のサイズ
                    const srcWidth = img.width;
                    const srcHeight = img.height;

                    // 16:9の計算
                    let targetWidth, targetHeight, offsetX = 0, offsetY = 0;

                    if (srcWidth / srcHeight > 16 / 9) {
                        // 横長すぎる場合
                        targetHeight = srcHeight;
                        targetWidth = (srcHeight * 16) / 9;
                        offsetX = (srcWidth - targetWidth) / 2;
                    } else {
                        // 縦長、または正方形の場合
                        targetWidth = srcWidth;
                        targetHeight = (srcWidth * 9) / 16;
                        offsetY = (srcHeight - targetHeight) / 2;
                    }

                    // Canvasのサイズを16:9に設定
                    canvas.width = targetWidth;
                    canvas.height = targetHeight;

                    // 画像を描画（中央切り抜き）
                    ctx.drawImage(img, offsetX, offsetY, targetWidth, targetHeight, 0, 0, targetWidth, targetHeight);

                    // クロップ後の画像をDataURLとして返す
                    resolve(canvas.toDataURL('image/png'));
                };
                img.onerror = reject;
                img.src = base64ImageUrl;
            });
        }

        // 生成ボタンクリック時の処理 (共通化)
        async function startGeneration(useEnhancement = false) {
            const files = fileInput.files;
            if (files.length === 0) {
                alert('プロンプトが書かれたテキストファイルを選択してください。');
                return;
            }

            // ベース画像が選択されている場合はBase64に変換する
            let base64BaseImage = null;
            let mimeTypeBaseImage = null;
            if (imageInput.files && imageInput.files.length > 0) {
                const imgFile = imageInput.files[0];
                mimeTypeBaseImage = imgFile.type;
                base64BaseImage = await new Promise((resolve, reject) => {
                    const reader = new FileReader();
                    reader.onload = () => resolve(reader.result.split(',')[1]);
                    reader.onerror = reject;
                    reader.readAsDataURL(imgFile);
                });
            }

            // UIと変数の初期化
            gallery.innerHTML = '';
            generatedZip = new JSZip();
            hasGeneratedImages = false;
            downloadZipBtn.disabled = true;
            generateBtn.disabled = true;
            generateEnhancedBtn.disabled = true;
            
            statusContainer.className = 'status-bar status-info';
            loader.style.display = 'block';

            let successCount = 0;
            let errorCount = 0;

            // 選択された各ファイルを順番に処理
            for (let i = 0; i < files.length; i++) {
                const file = files[i];
                statusText.textContent = `生成中... (${i + 1}/${files.length}) : ${file.name}`;
                
                let promptText = "";
                try {
                    // テキストファイルの内容を読み込む
                    promptText = await file.text();
                    if (!promptText.trim()) {
                        throw new Error("ファイルが空です");
                    }

                    // ★全体への追加指示を取得して結合
                    const globalPrompt = globalPromptInput.value.trim();
                    if (globalPrompt) {
                        promptText = promptText.trim() + ", " + globalPrompt;
                    }

                } catch (e) {
                    createErrorCard(file.name, "ファイルの読み込みに失敗しました");
                    errorCount++;
                    continue;
                }

                // 読み込み中のプレースホルダー（枠）を作成
                const cardId = `card-${Date.now()}-${i}`;
                createLoadingCard(cardId, file.name, promptText);

                try {
                    let finalPrompt = promptText;
                    
                    // LLMによるプロンプトの自動拡張機能
                    if (useEnhancement) {
                        statusText.textContent = `✨ プロンプト拡張中... (${i + 1}/${files.length}) : ${file.name}`;
                        finalPrompt = await enhancePrompt(promptText);
                        
                        // UIのプレビューテキストを拡張されたプロンプトに更新
                        const previewEl = document.querySelector(`#${cardId} .card-prompt-preview`);
                        if (previewEl) previewEl.textContent = `✨ ${finalPrompt}`;
                        
                        statusText.textContent = `画像生成中... (${i + 1}/${files.length}) : ${file.name}`;
                    }

                    // APIを呼び出して画像を生成 (正方形の可能性あり)
                    const originalImageUrl = await generateImage(finalPrompt, base64BaseImage, mimeTypeBaseImage);
                    
                    // キャンバスを使用して画像を16:9にクロップ（中央切り抜き）
                    const base64ImageUrl = await cropImageTo16x9(originalImageUrl);
                    
                    // 拡張子を除去して ".png" を付与 (例: sample.txt -> sample.png)
                    const imageFileName = file.name.replace(/\.[^/.]+$/, "") + ".png"; 
                    
                    // UIを生成された画像に更新
                    updateCardWithImage(cardId, imageFileName, base64ImageUrl, finalPrompt);
                    
                    // DataURLからBase64部分のみを抽出してZIPに追加
                    const base64Data = base64ImageUrl.split(',')[1];
                    generatedZip.file(imageFileName, base64Data, {base64: true});
                    
                    successCount++;
                    hasGeneratedImages = true;
                } catch (error) {
                    updateCardWithError(cardId, file.name, error.message);
                    errorCount++;
                }
            }

            // すべての処理が完了した時の後処理
            loader.style.display = 'none';
            generateBtn.disabled = false;
            generateEnhancedBtn.disabled = false;
            
            if (hasGeneratedImages) {
                downloadZipBtn.disabled = false;
                statusContainer.className = 'status-bar status-success';
                statusText.textContent = `完了: ${successCount}件成功, ${errorCount}件失敗`;
            } else {
                statusContainer.className = 'status-bar status-error';
                statusText.textContent = `処理終了: 画像を生成できませんでした（失敗 ${errorCount}件）`;
            }
        }

        generateBtn.addEventListener('click', () => startGeneration(false));
        generateEnhancedBtn.addEventListener('click', () => startGeneration(true));

        // ZIPダウンロードボタンクリック時の処理
        downloadZipBtn.addEventListener('click', async () => {
            if (!hasGeneratedImages) return;
            
            statusContainer.className = 'status-bar status-info';
            statusText.textContent = "ZIPファイルを準備中...";
            loader.style.display = 'block';
            downloadZipBtn.disabled = true;

            try {
                // JSZipで圧縮処理を実行
                const content = await generatedZip.generateAsync({type: "blob"});
                
                // ダウンロード用のリンクを生成してクリック
                const link = document.createElement('a');
                link.href = URL.createObjectURL(content);
                link.download = "generated_images.zip";
                link.click();
                URL.revokeObjectURL(link.href);
                
                statusContainer.className = 'status-bar status-success';
                statusText.textContent = "ZIPファイルのダウンロードを開始しました";
            } catch (error) {
                alert("ZIPの作成に失敗しました");
                statusContainer.className = 'status-bar status-error';
                statusText.textContent = "エラー: ZIP作成失敗";
            } finally {
                loader.style.display = 'none';
                downloadZipBtn.disabled = false;
            }
        });

        /**
         * 画像生成API呼び出し (指数バックオフ実装)
         * @param {string} prompt 元のプロンプト
         * @param {string} baseImage ベース画像のBase64データ(Image-to-Image用)
         * @param {string} mimeType ベース画像のMIMEタイプ
         */
        async function generateImage(prompt, baseImage = null, mimeType = null) {
            // 最大5回のリトライ。待機時間: 1s, 2s, 4s, 8s, 16s
            const delays = [1000, 2000, 4000, 8000, 16000];
            let attempt = 0;

            // ★ 16:9を強制するための指示文をプロンプトの末尾に自動追加
            const modifiedPrompt = prompt.trim() + ", 16:9 aspect ratio, thumbnail size, landscape";

            let fetchUrl = apiUrl;
            let payload = {};

            if (baseImage) {
                // Image-to-Imageモデルの仕様に合わせたリクエストを作成
                fetchUrl = `https://generativelanguage.googleapis.com/v1beta/models/gemini-2.5-flash-image-preview:generateContent?key=${apiKey}`;
                payload = {
                    contents: [
                        {
                            role: "user",
                            parts: [
                                { text: modifiedPrompt },
                                { inlineData: { mimeType: mimeType || "image/png", data: baseImage } }
                            ]
                        }
                    ],
                    generationConfig: { responseModalities: ["TEXT", "IMAGE"] }
                };
            } else {
                // Text-to-Imageモデルの仕様に合わせたリクエストを作成
                payload = {
                    instances: { prompt: modifiedPrompt },
                    // APIパラメータとしても16:9を指定
                    parameters: { sampleCount: 1, aspectRatio: "16:9" }
                };
            }

            while (attempt < 5) {
                try {
                    const response = await fetch(fetchUrl, {
                        method: 'POST',
                        headers: {
                            'Content-Type': 'application/json'
                        },
                        body: JSON.stringify(payload)
                    });

                    if (!response.ok) {
                        throw new Error(`HTTP error! status: ${response.status}`);
                    }

                    const data = await response.json();
                    
                    if (baseImage) {
                        // Image-to-Imageのレスポンス解析
                        const b64 = data.candidates?.[0]?.content?.parts?.find(p => p.inlineData)?.inlineData?.data;
                        if (b64) {
                            return `data:image/png;base64,${b64}`;
                        } else {
                            throw new Error('APIレスポンスに画像データが含まれていません');
                        }
                    } else {
                        // Text-to-Imageのレスポンス解析
                        if (data.predictions && data.predictions[0] && data.predictions[0].bytesBase64Encoded) {
                            return `data:image/png;base64,${data.predictions[0].bytesBase64Encoded}`;
                        } else {
                            throw new Error('APIレスポンスに画像データが含まれていません');
                        }
                    }

                } catch (error) {
                    if (attempt === 4) {
                        throw new Error('画像の生成に失敗しました。時間をおいて再度お試しください。');
                    }
                    // 指数バックオフ待機
                    await new Promise(resolve => setTimeout(resolve, delays[attempt]));
                    attempt++;
                }
            }
        }

        function escapeHtml(unsafe) {
            return (unsafe || "").replace(/&/g, "&amp;")
                .replace(/</g, "&lt;")
                .replace(/>/g, "&gt;")
                .replace(/"/g, "&quot;")
                .replace(/'/g, "&#039;");
