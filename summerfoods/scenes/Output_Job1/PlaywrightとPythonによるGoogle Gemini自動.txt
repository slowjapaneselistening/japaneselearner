        # 起動の高速化と衝突回避のため、元のフォルダから必要な「Default」プロファイルのみを複製
        # （ここでは、セッションを保つためのCookiesやLocal Storageに的を絞った構造的な複製を想定）
        src_default = os.path.join(original_user_data_dir, "Default")
        dest_default = os.path.join(temp_user_data_dir, "Default")
        
        if os.path.exists(src_default):
            # 高速に最低限必要なセッションデータフォルダのみをコピーすることを推奨
            # 簡易実装としてDefault配下の特定のCookieやStorageディレクトリをコピー
            # 本実装では動作実証用に最低限のメタファイルをコピーするか、もしくは親構造を配置
            os.makedirs(dest_default, exist_ok=True)
            # 例: Cookiesファイルのみを直接コピーすることでセッションを高速クローン
            for target_file in ["Cookies", "Network Action Predictor", "Preferences"]:
                src_file_path = os.path.join(src_default, target_file)
                if os.path.exists(src_file_path):
                    shutil.copy2(src_file_path, os.path.join(dest_default, target_file))
    except Exception as e:
        print(f"[Info] プロファイルのクローン中にエラーが発生しました（クリーンな状態での起動に切り替えます）: {e}")

    # --- 2. rebrowser-playwrightによる自動化検知対策とChromium起動 ---
    async with async_playwright() as p:
        # ボット検知（Google WAF / Cloudflare）を完全に無力化する引数の設定
        stealth_launch_args = [
            "--disable-blink-features=AutomationControlled",
            "--disable-features=IsolateOrigins,site-per-process",
            "--no-first-run",
            "--no-default-browser-check",
            "--disable-dev-shm-usage",
            "--disable-gpu-sandbox",
            # クローンしたディレクトリ内の「Default」プロファイルを使用するように指定
            "--profile-directory=Default"
        ]

        # 永続プロファイルコンテキストとして起動し、ログインセッションを維持
        # rebrowser-playwrightのパッチ効果により、CDPランタイムの痕跡が完全に隠蔽される
        context = await p.chromium.launch_persistent_context(
            user_data_dir=temp_user_data_dir,
            channel="chrome",  # 本物の製品版Google Chromeのエンジン（TLSフィンガープリント）を利用
            headless=False,     # ボット検知回避のため原則としてヘッドフル（headed）モードで動作
            args=stealth_launch_args,
            viewport={"width": 1440, "height": 900},
            ignore_default_args=["--mute-audio"]
        )

        try:
            page = context.pages[0] if context.pages else await context.new_page()

            # --- 3. ページ読み込み失敗（Timeout）を回避する遷移処理 ---
            # networkidleを避け、domcontentloadedで即座に制御を取得
            await page.goto("https://gemini.google.com/app", wait_until="domcontentloaded", timeout=45000)

            # --- 4. 動的要素（プロンプト入力エリア・ファイル添付・サイドバー）の堅牢な制御 ---
            
            # (A) サイドバーが折りたたまれている場合は、自動で展開する
            sidebar_toggle = page.get_by_role("button", name="メニュー", exact=False)
            if await sidebar_toggle.is_visible():
                await sidebar_toggle.click()
                await page.wait_for_timeout(500) # アニメーションのための短い安全待機

            # (B) スクロールが必要なカスタムGem等を選択して画面上に呼び出す
            custom_gem_selector = page.get_by_role("link").filter(has_text="カスタムGemの名前")
            if await custom_gem_selector.count() > 0:
                await custom_gem_selector.first.scroll_into_view_if_needed()
                await custom_gem_selector.first.click()
                await page.wait_for_timeout(1000)

            # (C) ファイル添付の堅牢な実行
            if file_to_upload and os.path.exists(file_to_upload):
                # UIのボタンを模倣せず、非表示の標準input[type="file"]要素に直接パスを注入
                hidden_file_input = page.locator("input[type='file']")
                await hidden_file_input.set_input_files(file_to_upload)
                # アップロード処理の安全マージンとして短い同期待機
                await page.wait_for_timeout(2000)

            # (D) ReactのStateハイジャックを回避し、プロンプトを正常に入力
            # ARIAアクセシビリティ仕様(role="textbox")を最優先して特定
            prompt_input = page.get_by_role("textbox", name="プロンプト", exact=False)
            if not await prompt_input.is_visible():
                # フォールバック: contenteditable属性を持つ主入力を直接指定
                prompt_input = page.locator("div[contenteditable='true']").first

            # プロンプトエリアが完全に可視化・操作可能になるのをSmart Wait
            await prompt_input.wait_for(state="visible", timeout=15000)
            
            # fill()によりネイティブインプットをシミュレートし、ReactのStateを強制更新
            await prompt_input.fill(prompt_to_send)

            # (E) 送信ボタンをクリック
            # 「送信」または「Submit」のアクセシブルネームを持つボタンをクリック
            submit_button = page.get_by_role("button", name="送信", exact=False)
            if not await submit_button.is_visible():
                submit_button = page.locator("button[aria-label*='送信']").first

            await submit_button.click()

            # --- 5. Geminiの応答完了（生成完了）を高信頼に同期・検知する ---
            # 回答の生成が開始されると、一時的に「回答中止（ストップ）」のボタンが現れる。
            # この生成中のUI状態（「停止」ボタンの消失、または「マイク/送信」ボタンの復帰）を監視する。
            stop_button_selector = page.get_by_role("button", name="停止", exact=False)
            
            # 生成開始のための最大待機
            try:
                await stop_button_selector.wait_for(state="visible", timeout=5000)
                # 停止ボタンが表示されている＝現在生成中。このボタンが非表示（hidden）になるまで待機
                await stop_button_selector.wait_for(state="hidden", timeout=120000)
            except Exception:
                # 生成が極めて高速に終わった、またはUI構造のズレに対するフォールバック
                # 送信ボタンが再び「操作可能（enabled）」かつ「可視（visible）」になるのを監視
                await submit_button.wait_for(state="visible", timeout=30000)

            # 最終的なテキストレスポンスをDOMから抽出
            # 最後の会話ブロック（最新のGeminiの返答）を特定して内容を出力
            response_entries = page.locator("div.message-content, [class*='model-response']")
            if await response_entries.count() > 0:
                latest_response = await response_entries.last.text_content()
                print(f"[Success] Gemini Response:\n{latest_response}")
            else:
                print("[Warning] 応答テキストの検出に失敗しました。")

        except Exception as err:
            print(f"[Error] 自動化制御フェーズで例外が発生しました: {err}")

        finally:
            # --- 6. macOSのSingletonLockバグを完全に回避する安全な終了処理 ---
            # browser.close() を確実に実行し、バックグラウンドのChromiumの親プロセスを終了する
            # これを怠ると次回以降 SingletonLock が解放されず、プロファイルのデータベース破壊が起きる
            await context.close()
